# Human-maintained source of truth for external dependencies.
# Bazel consumes a generated requirements.lock.txt via pip.parse; regenerate the lock after
# editing this file. Standard-library-only modules (e.g. the store's SQLite half) add nothing
# here.

# Store — cloud object-storage backend (lazy-imported; exercised only against a real bucket).
google-cloud-storage

# KB ingestion — the small fast model used for per-section extraction (lazy-imported; the
# hermetic tests inject the extraction call, so this is exercised only live).
anthropic

# File indexing — mesh math (measurements) and the software rasteriser for intended-geometry
# renders. The .3mf/G-code parsing itself is standard-library only.
numpy

# The Claude Agent SDK — hosts the in-process MCP servers and drives the agent loop (lazy-imported
# by the MCP wrapper and the orchestrator; exercised live, not in the hermetic suite).
claude-agent-sdk

# Voice transcription — the bundled faster-whisper `base` model transcribes uploaded clips
# server-side ([decisions.md 2026-07-23]). Lazy-imported by the Transcriber and depended on only by
# //printer_debugger:main / //:image, so `bazel test //...` never fetches the heavy wheels.
faster-whisper

# Procedures — the calibration catalog documents are YAML data loaded at startup.
PyYAML

# Web layer — the async framework, server, upload parsing, and (test-only) HTTP client.
fastapi
uvicorn
python-multipart
httpx
